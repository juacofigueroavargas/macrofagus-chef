function exportarAlimentosExcel() {
  const datos = alimentos.map(a => ({
    Alimento: a.nombre, Base: a.medida,
    Proteina: a.proteina, Carbohidratos: a.carbohidrato,
    Grasas: a.grasa, Kcal: calcularKcal(a)
  }));
  const hoja = XLSX.utils.json_to_sheet(datos);
  const libro = XLSX.utils.book_new();
  XLSX.utils.book_append_sheet(libro, hoja, "Alimentos");
  XLSX.writeFile(libro, "macrofagus_chef_alimentos.xlsx");
}

function importarAlimentosExcel(event) {
  const archivo = event.target.files[0];
  if (!archivo) return;
  const lector = new FileReader();
  lector.onload = function(e) {
    const data = new Uint8Array(e.target.result);
    const libro = XLSX.read(data, { type: "array" });
    const hoja = libro.Sheets[libro.SheetNames[0]];
    const filas = XLSX.utils.sheet_to_json(hoja);
    const nuevos = filas.map(f => ({
      nombre: f.Alimento || f.nombre || "",
      medida: f.Base || f.medida || "100 g",
      proteina: parseFloat(f.Proteina || f.proteina) || 0,
      carbohidrato: parseFloat(f.Carbohidratos || f.carbohidrato) || 0,
      grasa: parseFloat(f.Grasas || f.grasa) || 0
    })).filter(a => a.nombre !== "");
    if (nuevos.length === 0) { alert("El archivo no tiene alimentos válidos."); return; }
    alimentos = nuevos;
    mostrarAlimentos();
    renderComidas();
    alert("Alimentos importados correctamente.");
  };
  lector.readAsArrayBuffer(archivo);
}

function guardarAlimento() {
  const alimento = {
    nombre: document.getElementById('nombre').value.trim(),
    medida: document.getElementById('medida').value,
    proteina: parseFloat(document.getElementById('proteina').value) || 0,
    carbohidrato: parseFloat(document.getElementById('carbohidrato').value) || 0,
    grasa: parseFloat(document.getElementById('grasa').value) || 0,
    categoria: document.getElementById('categoria').value || 'cereales'
  };
  if (!alimento.nombre) { alert('Debes ingresar un alimento.'); return; }
  if (editandoIndex === null) {
    alimentos.push(alimento);
  } else {
    alimentos[editandoIndex] = alimento;
    editandoIndex = null;
    document.getElementById('btnGuardar').textContent = '+ Agregar alimento';
  }
  limpiarFormulario();
  mostrarAlimentos();
  renderComidas();
}

function toggleSeccion(cat) {
  seccionesColapsadas[cat] = !seccionesColapsadas[cat];

function mostrarAlimentos() {
  const contenedor = document.getElementById('tablaAlimentos');
  const buscar = document.getElementById('buscar').value.toLowerCase();

  const filtrados = alimentos
    .map((a, i) => ({ ...a, _idx: i }))
    .filter(a => a.nombre.toLowerCase().includes(buscar));

  // Update counters
  document.getElementById('contadorTotal').textContent = filtrados.length;

  // Pill counters per category
  const catsEl = document.getElementById('contadorCats');
  const porCat = {};
  filtrados.forEach(a => {
    const cat = a.categoria || 'sin_categoria';
    porCat[cat] = (porCat[cat] || 0) + 1;
  });

  catsEl.innerHTML = Object.entries(porCat).map(([cat, count]) => {
    const info = categorias[cat] || { nombre: cat, icon: '📦', color: '#888', bg: '#f0f0f0' };
    return `<span class="cat-pill" style="color:${info.color};background:${info.bg};border-color:${info.color}22">
      ${info.icon} ${info.nombre} <strong style="margin-left:2px">${count}</strong>
    </span>`;
  }).join('');

  // Render grouped sections
  let html = '';
  const catOrder = Object.keys(categorias);
  const otherCats = [...new Set(filtrados.map(a => a.categoria))].filter(c => !catOrder.includes(c));
  const allCats = [...catOrder, ...otherCats];

  allCats.forEach(cat => {
    const items = filtrados.filter(a => (a.categoria || 'sin_categoria') === cat);
    if (items.length === 0) return;

    const info = categorias[cat] || { nombre: cat, icon: '📦', color: '#888', bg: '#f0f0f0' };
    const collapsed = seccionesColapsadas[cat];

    html += `<div class="section-group">
      <div class="section-header" onclick="toggleSeccion('${cat}')">
        <div class="section-header-left">
          <div class="section-icon" style="background:${info.bg};color:${info.color}">${info.icon}</div>
          <span class="section-name" style="color:${info.color}">${info.nombre}</span>
          <span class="section-count" style="background:${info.bg};color:${info.color}">${items.length}</span>
        </div>
        <span class="section-toggle${collapsed ? ' collapsed' : ''}">▾</span>
      </div>
    `;

    if (!collapsed) {
      html += `<div class="section-table-wrap"><table>
        <thead><tr>
          <th>Alimento</th><th>Base</th><th>Proteína</th>
          <th>Carbohidratos</th><th>Grasas</th><th>Kcal</th><th>Acciones</th>
        </tr></thead>
        <tbody>`;
      items.forEach(a => {
        html += `<tr>
          <td><strong>${a.nombre}</strong></td>
          <td>${a.medida}</td>
          <td>${a.proteina} g</td>
          <td>${a.carbohidrato} g</td>
          <td>${a.grasa} g</td>
          <td>${calcularKcal(a).toFixed(0)} kcal</td>
          <td><div class="action-cell">
            <button class="btn-amber" onclick="editarAlimento(${a._idx})">Editar</button>
            <button class="btn-red" onclick="eliminarAlimento(${a._idx})">Eliminar</button>
          </div></td>
        </tr>`;
      });
      html += `</tbody></table></div>`;
    }

    html += `</div>`;
  });

  if (html === '') {
    html = `<div class="empty-section">No se encontraron alimentos.</div>`;
  }

  contenedor.innerHTML = html;
}

function calcularObjetivoComida(comida) {
  const objP = parseFloat(document.getElementById('macroProteina').value) || 0;
  const objC = parseFloat(document.getElementById('macroCarbo').value) || 0;
  const objG = parseFloat(document.getElementById('macroGrasa').value) || 0;
  const pcts = { desayuno:30, colacion1:10, almuerzo:30, colacion2:10, cena:20 };
  const f = pcts[comida] / 100;
  const proteina = objP * f, carbohidrato = objC * f, grasa = objG * f;
  return { porcentaje: pcts[comida], proteina, carbohidrato, grasa,
    kcal: proteina*4 + carbohidrato*4 + grasa*9 };
}

function calcularConsumidoComida(comida) {
  let total = { proteina:0, carbohidrato:0, grasa:0, kcal:0 };
  comidas[comida].forEach(item => {
    total.proteina += item.proteina;
    total.carbohidrato += item.carbohidrato;
    total.grasa += item.grasa;
    total.kcal += item.kcal;
  });
  return total;
}

      </div>

function agregarAlimentoDeseado(comida, index) {

  document.getElementById(`lista-sugerencia-${comida}`).style.display = 'none';

function renderAlimentosDeseados(comida) {
  if (alimentosDeseados[comida].length === 0)
    return `<small style="color:var(--muted);font-size:12px">Agrega alimentos para generar sugerencias.</small>`;
  return alimentosDeseados[comida].map(i =>
    `<button class="chip" onclick="quitarAlimentoDeseado('${comida}',${i})">${alimentos[i].nombre} ✕</button>`
  ).join('');
}

function quitarAlimentoDeseado(comida, index) {
  alimentosDeseados[comida] = alimentosDeseados[comida].filter(i => i !== index);
  document.getElementById(`alimentos-deseados-${comida}`).innerHTML = renderAlimentosDeseados(comida);
}

function cantidadAleatoria(medida) {
  if (medida === '1 scoop') return Math.ceil(Math.random() * 2);
  if (medida === '1 unidad') return Math.ceil(Math.random() * 4);
  if (medida === '1 litro') return Math.round((50 + Math.random() * 450) / 10) * 10;
  return Math.round((20 + Math.random() * 280) / 5) * 5;
}

function calcularMacrosPorCantidad(alimento, cantidad) {
  let factor = cantidad / 100;
  if (alimento.medida === '1 litro') factor = cantidad / 1000;
  if (alimento.medida === '1 scoop' || alimento.medida === '1 unidad') factor = cantidad;
  const p = alimento.proteina * factor, c = alimento.carbohidrato * factor, g = alimento.grasa * factor;
  return { proteina: p, carbohidrato: c, grasa: g, kcal: p*4 + c*4 + g*9 };
}

function generarSugerenciaComida(comida) {
  const objetivo = calcularObjetivoComida(comida);
  const seleccionados = alimentosDeseados[comida];
  if (seleccionados.length === 0) { alert('Primero elige alimentos para esta comida.'); return; }
  let mejorOpcion = null, mejorPuntaje = Infinity;
  for (let i = 0; i < 7000; i++) {
    let opcion = [], total = { proteina:0, carbohidrato:0, grasa:0, kcal:0 };
    seleccionados.forEach(index => {
      const a = alimentos[index];
      const cantidad = cantidadAleatoria(a.medida);
      const macros = calcularMacrosPorCantidad(a, cantidad);
      opcion.push({ index, nombre:a.nombre, medida:a.medida, cantidad, ...macros });
      total.proteina += macros.proteina;
      total.carbohidrato += macros.carbohidrato;
      total.grasa += macros.grasa;
      total.kcal += macros.kcal;
    });
    const puntaje =
      Math.abs(total.proteina - objetivo.proteina) * 4 +
      Math.abs(total.carbohidrato - objetivo.carbohidrato) * 3 +
      Math.abs(total.grasa - objetivo.grasa) * 6 +
      Math.abs(total.kcal - objetivo.kcal) * 0.05;
    if (puntaje < mejorPuntaje) { mejorPuntaje = puntaje; mejorOpcion = { opcion, total }; }
  }
  mostrarSugerencia(comida, mejorOpcion, objetivo);
}

function mostrarSugerencia(comida, resultado, objetivo) {
  const contenedor = document.getElementById(`sugerencia-${comida}`);
  contenedor.style.display = 'block';
  let html = `
    <div style="font-size:12px;font-weight:600;text-transform:uppercase;letter-spacing:.06em;color:var(--muted);margin-bottom:12px">Opción sugerida</div>
    <div class="table-wrap">
    <table>
      <thead><tr><th>Alimento</th><th>Cantidad</th><th>P</th><th>C</th><th>G</th><th>Kcal</th></tr></thead>
      <tbody>
  `;
  resultado.opcion.forEach(item => {
    html += `<tr>
      <td>${item.nombre}</td>
      <td>${item.cantidad} ${obtenerTextoCantidad(item.medida)}</td>
      <td>${item.proteina.toFixed(1)}</td>
      <td>${item.carbohidrato.toFixed(1)}</td>
      <td>${item.grasa.toFixed(1)}</td>
      <td>${item.kcal.toFixed(0)}</td>
    </tr>`;
  });
  html += `</tbody></table></div>
    <div style="margin-top:12px;font-size:13px;color:var(--muted)">
      Total: P ${resultado.total.proteina.toFixed(1)}/${objetivo.proteina.toFixed(1)}g ·
      C ${resultado.total.carbohidrato.toFixed(1)}/${objetivo.carbohidrato.toFixed(1)}g ·
      G ${resultado.total.grasa.toFixed(1)}/${objetivo.grasa.toFixed(1)}g ·
      ${resultado.total.kcal.toFixed(0)}/${objetivo.kcal.toFixed(0)} kcal
    </div>
    <div style="margin-top:12px">
      <button class="btn-green" onclick="usarSugerencia('${comida}')">✓ Usar esta comida</button>
    </div>
  `;
  window[`ultimaSugerencia_${comida}`] = resultado.opcion;
  contenedor.innerHTML = html;
}

function usarSugerencia(comida) {
  const sugerencia = window[`ultimaSugerencia_${comida}`];
  if (!sugerencia) return;
  sugerencia.forEach(item => {
    comidas[comida].push({
      nombre: item.nombre, medida: item.medida, cantidad: item.cantidad,
      proteina: item.proteina, carbohidrato: item.carbohidrato,
      grasa: item.grasa, kcal: item.kcal
    });
  });
  renderComidas();
}

function agregarAComida(comida) {
  const indexAlimento = document.getElementById(`alimento-${comida}`).value;
  const cantidad = parseFloat(document.getElementById(`cantidad-${comida}`).value) || 0;
  if (indexAlimento === '') { alert('Selecciona un alimento.'); return; }
  if (cantidad <= 0) { alert('Ingresa una cantidad válida.'); return; }
  const a = alimentos[indexAlimento];
  const macros = calcularMacrosPorCantidad(a, cantidad);
  comidas[comida].push({ nombre:a.nombre, medida:a.medida, cantidad, ...macros });
  renderComidas();
}

function renderItemsComida(comida) {
  if (comidas[comida].length === 0) return `<tr><td colspan="7" style="color:var(--muted);text-align:center;padding:20px;font-size:13px">Sin alimentos agregados aún.</td></tr>`;
  return comidas[comida].map((item, index) => `
    <tr>
      <td>${item.nombre}</td>
      <td>${item.cantidad} ${obtenerTextoCantidad(item.medida)}</td>
      <td>${item.proteina.toFixed(1)}</td>
      <td>${item.carbohidrato.toFixed(1)}</td>
      <td>${item.grasa.toFixed(1)}</td>
      <td>${item.kcal.toFixed(0)}</td>
      <td><button class="btn-red" style="padding:4px 10px;font-size:12px" onclick="eliminarDeComida('${comida}',${index})">✕</button></td>
    </tr>
  `).join('');
}

function eliminarDeComida(comida, index) {
  comidas[comida].splice(index, 1);
  renderComidas();
}

function obtenerTextoCantidad(medida) {
  if (medida === '1 scoop') return 'scoop';
  if (medida === '1 unidad') return 'unid.';
  if (medida === '1 litro') return 'ml';
  return 'g';
}

function actualizarTotales() {
  let totalP = 0, totalC = 0, totalG = 0, totalK = 0;
  Object.values(comidas).forEach(lista => {
    lista.forEach(item => {
      totalP += item.proteina; totalC += item.carbohidrato;
      totalG += item.grasa; totalK += item.kcal;
    });
  });
  const objP = parseFloat(document.getElementById('macroProteina').value) || 0;
  const objC = parseFloat(document.getElementById('macroCarbo').value) || 0;
  const objG = parseFloat(document.getElementById('macroGrasa').value) || 0;
  const kcalObj = objP*4 + objC*4 + objG*9;
  document.getElementById('resProteina').textContent = `${totalP.toFixed(1)} / ${objP} g`;
  document.getElementById('resCarbo').textContent = `${totalC.toFixed(1)} / ${objC} g`;
  document.getElementById('resGrasa').textContent = `${totalG.toFixed(1)} / ${objG} g`;
  document.getElementById('resKcal').textContent = `${totalK.toFixed(0)} / ${kcalObj.toFixed(0)}`;
}
